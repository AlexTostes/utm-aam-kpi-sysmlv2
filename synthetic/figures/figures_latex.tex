\begin{figure}[!t]

\input{figures/fig_E2a_demand.tex}
\caption{Sensitivity of the prioritized KPI candidates to demand under the synthetic base-case
specification ($\tau = 60$\,s and $\Delta = 5$\,min for UTM; $\tau = 5$\,min and $\Delta = 10$\,min

\begin{figure}[!t]
\centering
\input{figures/fig_E2b_tau_delta.tex}
\caption{Sensitivity to the measurement specification. The same base-case logs are re-measured with each
value of the tolerance $\tau$ (predictability candidates) and of the shift window $\Delta$ (capacity
candidates); the dotted line marks the proposed value. Means of 30 replications with 95\% confidence
intervals. In the AAM base case, a tolerance of 10\,min or more drives AAM-PRED-01 towards saturation.}
\label{fig:e2b}
\end{figure}

\begin{figure}[!t]
\centering
\input{figures/fig_E4_tradeoff.tex}
\caption{Capacity--predictability trade-off under an operational lever: (a) the separation buffer between
UTM operational intents sharing an airspace volume (0--240\,s); (b) the schedule buffer before the next AAM
departure (0--10\,min). Labels mark the end values of each sweep. Means of 30 replications with 95\%
confidence intervals on both axes.}
\label{fig:e4}
\end{figure}

\begin{figure}[!t]
\centering
\input{figures/fig_E5_federation.tex}
